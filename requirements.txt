# Quality-Aware Self-Correcting Speech Translation on an Edge Device
# Target: NVIDIA Jetson Nano 4GB, JetPack 4.x (Ubuntu 18.04, aarch64)
# Python: 3.8 (installed alongside the system Python 3.6)
# Inference is CPU-only; no CUDA build of PyTorch is required.
#
# Install:  python3.8 -m venv .venv && source .venv/bin/activate
#           pip install --upgrade pip && pip install -r requirements.txt

# ===== Core ML / NLP =====
torch==2.2.2               # CPU inference (aarch64 wheel from PyPI)
transformers==4.20.0       # Opus-MT (MarianMT) + mBERT QE; tested stack is pinned <=4.30
sentencepiece==0.2.0       # MarianTokenizer
sacremoses                 # MarianTokenizer dependency
openai-whisper             # Whisper tiny ASR (requires the ffmpeg binary, see below)

# ===== Evaluation =====
sacrebleu==1.5.1           # BLEU / ChrF. Scripts assume the 1.x API where corpus_chrf().score is 0-1
                           # and multiply by 100; sacrebleu>=2.0 already returns 0-100.
jiwer                      # WER for Whisper on FLEURS (phase4_baseline.py)

# ===== Audio =====
numpy==1.24.4              # last release supporting Python 3.8
scipy==1.10.1              # last release supporting Python 3.8
sounddevice==0.5.3         # microphone capture (needs PortAudio)
soundfile                  # WAV I/O for the recording -> ASR temp-file bridge (needs libsndfile)
noisereduce==3.0.3         # spectral-gating noise reduction (optional at runtime)

# ===== System dependencies (apt, not pip) =====
# sudo apt-get install -y python3.8 python3.8-dev python3.8-venv \
#     espeak-ng alsa-utils ffmpeg libsndfile1 portaudio19-dev libportaudio2
#   espeak-ng  - TTS engine (called via subprocess)
#   alsa-utils - provides `aplay` for TTS playback
#   ffmpeg     - required by openai-whisper to read audio files
#   libsndfile1 / portaudio - backends for soundfile / sounddevice
#
# ===== Not included =====
# COMET scoring, GER and bootstrap significance tests were run off-device
# on Google Colab and are not part of the on-device requirements.
